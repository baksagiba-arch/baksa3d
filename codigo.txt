// Banco de dados do app Baksa.3d (uma aba por coleção: id | dados | atualizado)
function aba_(col) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(col);
  if (!sh) { sh = ss.insertSheet(col); sh.appendRow(["id", "dados", "atualizado"]); }
  return sh;
}
function saida_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  var cols = (e.parameter.cols || "filamentos,pedidos,cotacoes").split(",");
  var res = {};
  cols.forEach(function (col) {
    var v = aba_(col).getDataRange().getValues();
    res[col] = [];
    for (var i = 1; i < v.length; i++) {
      if (!v[i][0]) continue;
      try { var d = JSON.parse(v[i][1]); d.id = String(v[i][0]); res[col].push(d); } catch (x) {}
    }
  });
  return saida_({ ok: true, dados: res });
}
function doPost(e) {
  var lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    var b = JSON.parse(e.postData.contents);
    var sh = aba_(b.col);
    var ids = sh.getRange(1, 1, sh.getLastRow(), 1).getValues().map(function (r) { return String(r[0]); });
    var linha = ids.indexOf(String(b.id)) + 1;
    if (b.op === "apagar") { if (linha > 1) sh.deleteRow(linha); }
    else {
      var reg = [String(b.id), JSON.stringify(b.dados), new Date()];
      if (linha > 1) sh.getRange(linha, 1, 1, 3).setValues([reg]); else sh.appendRow(reg);
    }
    return saida_({ ok: true });
  } finally { lock.releaseLock(); }
}
